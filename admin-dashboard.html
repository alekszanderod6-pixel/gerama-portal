GERAMA Connect – Admin Controls

Monitor chats, statuses, group chats and user activity on GERAMA Connect. Block or flag suspicious users.

New Connect columns needed: To enable message editing, replies, reactions, and image messages, run the SQL migration in Supabase. 
-- Run in Supabase SQL Editor:
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS media_type text;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS media_url text;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS reaction text;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS reply_to_id uuid;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS reply_to_text text;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS is_edited boolean DEFAULT false;
ALTER TABLE connect_messages ADD COLUMN IF NOT EXISTS is_deleted boolean DEFAULT false;

-- Allow authenticated users to update messages (for read receipts, edits):
DROP POLICY IF EXISTS "Auth update messages" ON connect_messages;
CREATE POLICY "Auth update messages" ON connect_messages FOR UPDATE
  USING (sender_email = auth.email() OR recipient_email = auth.email());

-- Enable realtime for live chat (if not already):
ALTER PUBLICATION supabase_realtime ADD TABLE connect_messages;

–️ Mall Tables Not Found – Run Setup SQL

The mall_products table doesn't exist yet in your Supabase project. Copy the SQL below and run it in your Supabase – SQL Editor.

-- GERAMA Mall Tables Setup
CREATE TABLE IF NOT EXISTS mall_products (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  name text NOT NULL,
  category text NOT NULL,
  price numeric NOT NULL,
  original_price numeric,
  description text,
  image_url text,
  seller_name text,
  seller_whatsapp text,
  seller_id text,
  status text DEFAULT 'approved',
  colours text DEFAULT '[]',
  colour_names text DEFAULT '[]',
  sizes text DEFAULT '[]',
  stock integer DEFAULT 0,
  verified boolean DEFAULT true,
  featured boolean DEFAULT false,
  badge text,
  extra_images text DEFAULT '[]',
  created_at timestamptz DEFAULT now()
);
ALTER TABLE mall_products ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read mall_products" ON mall_products FOR SELECT USING (true);
CREATE POLICY "Anon all mall_products" ON mall_products FOR ALL USING (true) WITH CHECK (true);

CREATE TABLE IF NOT EXISTS mall_seller_requests (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  product_name text NOT NULL,
  category text,
  price numeric,
  original_price numeric,
  description text,
  image_url text,
  extra_images text,
  seller_name text,
  seller_whatsapp text,
  seller_id text,
  colours text,
  sizes text,
  stock integer DEFAULT 0,
  status text DEFAULT 'pending',
  submitted_at timestamptz DEFAULT now()
);
ALTER TABLE mall_seller_requests ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read mall_seller_requests" ON mall_seller_requests FOR SELECT USING (true);
CREATE POLICY "Anon all mall_seller_requests" ON mall_seller_requests FOR ALL USING (true) WITH CHECK (true);

CREATE TABLE IF NOT EXISTS mall_orders (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id text,
  product_name text,
  seller_id text,
  seller_whatsapp text,
  buyer_name text,
  buyer_phone text,
  colour text,
  size text,
  quantity integer DEFAULT 1,
  total_price numeric,
  delivery_location text,
  delivery_timeframe text,
  status text DEFAULT 'pending',
  payment_confirmed boolean DEFAULT false,
  item_delivered boolean DEFAULT false,
  ordered_at timestamptz DEFAULT now()
);
ALTER TABLE mall_orders ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read mall_orders" ON mall_orders FOR SELECT USING (true);
CREATE POLICY "Anon all mall_orders" ON mall_orders FOR ALL USING (true) WITH CHECK (true);

-- T-Shirt interest registrations
CREATE TABLE IF NOT EXISTS tshirt_interest (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_email text NOT NULL,
  full_name text,
  phone text,
  size_preference text,
  colour_preference text,
  registered_at timestamptz DEFAULT now(),
  UNIQUE(user_email)
);
ALTER TABLE tshirt_interest ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read tshirt_interest" ON tshirt_interest FOR SELECT USING (true);
CREATE POLICY "Anon all tshirt_interest" ON tshirt_interest FOR ALL USING (true) WITH CHECK (true);
